-- 1. CLEANUP OLD TABLES & POLICIES DROP TABLE IF EXISTS public.submissions CASCADE; DROP TABLE IF EXISTS public.profiles CASCADE; -- 2. CREATE USER PROFILES TABLE (Tied to Supabase Auth) CREATE TABLE public.profiles ( id UUID PRIMARY KEY REFERENCES auth.users(id) ON DELETE CASCADE, full_name TEXT NOT NULL, affiliation TEXT, role TEXT DEFAULT 'Author', -- Author, Reviewer, Editor created_at TIMESTAMPTZ DEFAULT NOW() ); -- 3. CREATE COMPREHENSIVE MANUSCRIPTS & SUBMISSIONS TABLE CREATE TABLE public.submissions ( id BIGINT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY, manuscript_id TEXT UNIQUE NOT NULL, user_id UUID REFERENCES auth.users(id) ON DELETE SET NULL, author_name TEXT NOT NULL, author_email TEXT NOT NULL, affiliation TEXT NOT NULL, title TEXT NOT NULL, domain TEXT NOT NULL, manuscript_type TEXT NOT NULL, abstract TEXT NOT NULL, keywords TEXT, coauthors JSONB DEFAULT '[]'::jsonb, file_url TEXT NOT NULL, status TEXT DEFAULT 'Submitted', -- Submitted, Under Review, Accepted, Published, Rejected volume INT DEFAULT 1, issue INT DEFAULT 1, doi TEXT, created_at TIMESTAMPTZ DEFAULT NOW() ); -- 4. ENABLE ROW LEVEL SECURITY (RLS) ALTER TABLE public.profiles ENABLE ROW LEVEL SECURITY; ALTER TABLE public.submissions ENABLE ROW LEVEL SECURITY; -- 5. RLS POLICIES FOR PROFILES CREATE POLICY "Public Profiles Access" ON public.profiles FOR SELECT USING (true); CREATE POLICY "Users can insert profile" ON public.profiles FOR INSERT WITH CHECK (auth.uid() = id); CREATE POLICY "Users can update own profile" ON public.profiles FOR UPDATE USING (auth.uid() = id); -- 6. RLS POLICIES FOR SUBMISSIONS -- Everyone (including anonymous visitors) can view published articles CREATE POLICY "Public view published submissions" ON public.submissions FOR SELECT USING (status = 'Published' OR auth.uid() = user_id); -- Logged in authors can insert submissions CREATE POLICY "Authenticated users submit manuscript" ON public.submissions FOR INSERT WITH CHECK (true); -- Authors can update their own pending submissions CREATE POLICY "Authors update own submissions" ON public.submissions FOR UPDATE USING (auth.uid() = user_id); -- 7. ENSURE STORAGE BUCKET EXISTS WITH PUBLIC ACCESS INSERT INTO storage.buckets (id, name, public) VALUES ('manuscripts', 'manuscripts', true) ON CONFLICT (id) DO UPDATE SET public = true; CREATE POLICY "Public storage upload" ON storage.objects FOR INSERT WITH CHECK (bucket_id = 'manuscripts'); CREATE POLICY "Public storage read" ON storage.objects FOR SELECT USING (bucket_id = 'manuscripts'); -- 8.
